	rain/*.cpp
	rain/*.h
	swe/*.cpp
	swe/*.h
	swe/fv1/*.cpp
	swe/fv1/*.h
	swe/dg2/*.cpp
	swe/dg2/*.h
)

# 头文件目录（整合 VS 下的设置）
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base/WetSpaInterface2
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base/bmps
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base/module_setting
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base/data
    /data/user/xiaodw/software/WISE/seims/src/seims_main/base/common_algorithm
    /data/user/xiaodw/software/WISE/seims/src/ccgl/src
    /data/user/xiaodw/software/gdal/include
    /data/user/xiaodw/software/mongo-c-driver/include/libbson-1.0
    /data/user/xiaodw/software/mongo-c-driver/include/libmongoc-1.0

# 添加库

# 链接基础库

    module_setting
    data
    bmps

    mongoc-1.0
    netcdf

    USE_GDAL
